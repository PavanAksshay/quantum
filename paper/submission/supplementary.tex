\subsection{Preprocessing and Text Standardization}
All raw communications were parsed to extract standard natural language text while stripping transmission artifacts:
\begin{itemize}
    \item \textbf{Email Corpora (CEAS 2008, MeAJOR)}: Parsed RFC 2822 email bodies and subject lines. HTML tags, entity references (\texttt{\&nbsp;}, \texttt{\&amp;}), MIME boundary strings, and transport headers (e.g., \texttt{Received:}, \texttt{X-Spam-Status:}) were removed. The processed input was formed as \texttt{f"\{subject\} \{body\}"}.
    \item \textbf{SMS Spam}: Raw text normalized by stripping Unicode control characters and extraneous whitespace.
\end{itemize}

\section{Section S2: Frozen Split Partitions and Sampling Hygiene}
\label{sec:supp_splits}

To eliminate data leakage, split partitions were deterministically computed and saved to disk.
\begin{enumerate}
    \item \textbf{Split Generation Hash}: Sample IDs were partitioned using deterministic pseudo-random shuffling with a fixed cryptographic seed to generate \texttt{train\_sample\_ids.csv}, \texttt{validation\_sample\_ids.csv}, and \texttt{test\_sample\_ids.csv}.
    \item \textbf{Deterministic Partitions}: For all in-distribution evaluations, the test set sample indices remain identical across all 10 experimental seeds. The seed parameter governs downstream stochasticity in TruncatedSVD initialization and SVC dual coordinate descent solvers.
    \item \textbf{Stratification}: All split generations enforce exact class-prevalence preservation across training, validation, and testing partitions.
\end{enumerate}

\section{Section S3: Deduplication and Data Quality Diagnostics}
\label{sec:supp_dedup}

Prior to benchmark execution, corpora were audited for identical and near-duplicate records:
\begin{itemize}
    \item \textbf{Exact Duplicates}: Within the SMS Spam collection, 403 exact duplicate messages were identified (predominantly standard carrier notifications and chain spams); these were retained in the standard benchmark to reflect operational message frequency distribution.
    \item \textbf{Null Content Handling}: Emails with empty subject lines and empty bodies following HTML stripping were discarded ($<0.01\%$ of total records).
    \item \textbf{Label Integrity}: 100\% of analyzed records possessed valid binary ground-truth labels.
\end{itemize}

\section{Section S7: Statistical Test Protocols and Null Distributions}
\label{sec:supp_stats}

\begin{enumerate}
    \item \textbf{Bootstrap Procedure}: For paired test differences $\Delta_s = \text{F1}_s^{\text{Q}} - \text{F1}_s^{\text{RBF}}$ across seeds $s \in \{1, \dots, 10\}$, $B = 10,000$ bootstrap vectors $\mathbf{\Delta}^*$ were generated by uniform sampling with replacement. The $95\%$ confidence interval was computed as $[q_{0.025}(\bar{\mathbf{\Delta}}^*), q_{0.975}(\bar{\mathbf{\Delta}}^*)]$.
    \item \textbf{Permutation Procedure}: Under $H_0: \mathbb{E}[\Delta] = 0$, $M = 10,000$ permuted mean differences were constructed via independent Rademacher sign variables $\sigma_s \in \{-1, +1\}$:
    \begin{equation}
    \bar{\Delta}^{(m)} = \frac{1}{10} \sum_{s=1}^{10} \sigma_s^{(m)} \Delta_s
    \end{equation}
    The two-sided empirical $p$-value was computed as $\frac{1}{M} \sum_{m=1}^M \mathbb{I}(|\bar{\Delta}^{(m)}| \ge |\bar{\Delta}^{\text{obs}}|)$.
\end{enumerate}

\section{Section S8: Feature Space Geometry and Alignment Derivations}
\label{sec:supp_geometry}

\subsection{Metric Definitions}
\begin{itemize}
    \item \textbf{Kernel-Target Alignment}:
    \begin{equation}
    A(K, \mathbf{y}) = \frac{\langle K, \mathbf{y}\mathbf{y}^T \rangle_F}{\|K\|_F \|\mathbf{y}\mathbf{y}^T\|_F} = \frac{\sum_{i, j} K_{ij} y_i y_j}{\sqrt{\sum_{i, j} K_{ij}^2} \sqrt{\sum_{i, j} (y_i y_j)^2}}
    \end{equation}
    \item \textbf{Single-State von Neumann Entropy}: For pure state $| \psi(\mathbf{x}) \rangle$, single-qubit reduced density matrix $\rho_1 = \text{Tr}_{\bar{1}}(|\psi\rangle\langle\psi|)$ yields entropy:
    \begin{equation}
    S(\rho_1) = -\text{Tr}(\rho_1 \log_2 \rho_1)
    \end{equation}
    \item \textbf{Pairwise Kernel Diversity}:
    \begin{equation}
    D(K) = \text{Var}_{i \neq j}(K_{ij}) = \frac{1}{N(N-1)} \sum_{i \neq j} (K_{ij} - \bar{K}_{\text{off}})^2
    \end{equation}
\end{itemize}
